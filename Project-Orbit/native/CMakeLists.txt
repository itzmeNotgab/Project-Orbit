cmake_minimum_required(VERSION 3.18)
project(orbit_native CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(orbit_core INTERFACE)
target_include_directories(orbit_core INTERFACE include)
# Shared runtime for the Android mod loader (built with the NDK from the app/launcher side).
add_library(orbit_runtime STATIC src/orbit_capi.cpp)
target_link_libraries(orbit_runtime PUBLIC orbit_core)
if(NOT ANDROID)
  enable_testing()
  add_executable(orbit_core_tests tests/test_core.cpp)
  target_link_libraries(orbit_core_tests orbit_core)
  add_executable(orbit_capi_tests tests/test_capi.cpp)
  target_link_libraries(orbit_capi_tests orbit_runtime)
  foreach(t orbit_core_tests orbit_capi_tests)
    target_compile_options(${t} PRIVATE -Wall -Wextra)
  endforeach()
  add_test(NAME core COMMAND orbit_core_tests)
  add_test(NAME capi COMMAND orbit_capi_tests)
endif()
